# OpenScreen MCP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A local stdio MCP server that lets Claude understand an OpenScreen recording (cursor clicks/pauses, key frames, scene changes, speech) and edit the app's `.openscreen` projects (zoom, cut, text, blur, speed, update, remove, undo) safely.

**Architecture:** TypeScript on Node. Pure edit functions transform the raw project JSON using a time-mapping module (timeline seconds ↔ clip-anchored source seconds). A store validates every result against OpenScreen's own vendored Zod schema, backs up, and writes atomically, refusing while the app runs. Analysis helpers wrap ffmpeg, whisper-cli and the recording's cursor file. A thin tool layer exposes it all through `@modelcontextprotocol/sdk`.

**Tech Stack:** Node ≥ 22 (ESM), TypeScript ~5.9, `@modelcontextprotocol/sdk` ^1.30.1, `zod` ^4.1.12, Vitest **4.0.18** (pinned: npm 10.9.2 crashes with `edgesOut` resolving `vitest` 4.1.x peers), esbuild, ffmpeg/ffprobe (Homebrew), whisper-cli (`brew install whisper-cpp`, optional).

**Spec:** `docs/superpowers/specs/2026-09-25-openscreen-mcp-design.md`

## Global Constraints

- Node ≥ 22, `"type": "module"`, TypeScript `strict`. Log only to stderr (`console.error`); stdout is the MCP channel.
- `zod` major version 4 (upstream pins `^4.1.12`). `@modelcontextprotocol/sdk` ^1.30.1.
- `src/vendor/openscreen-schema.ts` is generated by `scripts/vendor-schema.sh` from `getopenscreen/openscreen` tag **`v1.12.2`** and is never hand-edited.
- Only `schemaVersion: 7` projects are read or written.
- The schema is a **validation gate only**. Write `JSON.stringify(editedRawDoc, null, 2)`, never the parse output (zod strips unknown keys). Every region the server creates spells out all of its fields.
- Every edit (and undo) refuses while a process named exactly `Openscreen` is running; reads never check.
- Every tool input/output time is **timeline seconds** (RAW ruler: cut sections still occupy space).
- Backups: `<dataDir>/mcp-backups/<projectId>/`, newest 20 kept, `journal.json` alongside.
- IDs are `${prefix}_${uuid}` with prefixes `zoom`, `trim`, `annotation`, `speed`.
- Tests never touch `~/Library/Application Support/openscreen`: they set `OPENSCREEN_DATA_DIR` to a temp dir (or construct `Paths` from one).
- Overlap: zoom↔zoom, speed↔speed, text↔text, blur↔blur rejected; cuts may overlap.
- Every commit message ends with the trailer `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```ts
import { readFileSync } from "node:fs";
import type { Clip, Doc } from "../../src/project/types";

const FIXTURE = new URL("../fixtures/real-project.openscreen", import.meta.url);

/** A project exactly as OpenScreen 1.12.2 saved it. */
export function realDoc(): Doc {
	return JSON.parse(readFileSync(FIXTURE, "utf8")) as Doc;
}

export const ASSET_ID = realDoc().assets[0]!.id;

export function clip(
	id: string,
	sourceStartSec: number,
	sourceEndSec: number,
	timelineStartSec: number,
): Clip {
	return {
		id,
		assetId: ASSET_ID,
		sourceStartSec,
		sourceEndSec,
		timelineStartSec,
		timelineEndSec: timelineStartSec + (sourceEndSec - sourceStartSec),
		wordRefs: [],
		origin: "user",
		reason: "",
	};
}

function withClips(clips: Clip[]): Doc {
	const doc = realDoc();
	doc.timeline.clips = clips;
	doc.timeline.trimRanges = [];
	doc.zoomRanges = [];
	doc.annotations = [];
	doc.legacyEditor = { ...(doc.legacyEditor ?? {}), speedRegions: [] };
	return doc;
}

/** One clip: timeline 0–30 s = source 0–30 s. */
export const singleClipDoc = (): Doc => withClips([clip("clip_a", 0, 30, 0)]);

/** Two clips of one recording: timeline 0–10 = source 0–10, timeline 10–20 = source 20–30. */
export const twoClipDoc = (): Doc =>
	withClips([clip("clip_a", 0, 10, 0), clip("clip_b", 20, 30, 10)]);
```

```ts
// Stand-ins for the three helpers the vendored OpenScreen schema imports.
// The server only parses schemaVersion 7 documents, so the migration code that
// calls `anchorRegionsWithDerivedMs` never runs here.

export function createId(prefix: string): string {
	return `${prefix}_${crypto.randomUUID()}`;
}

function greatestCommonDivisor(a: number, b: number): number {
	return b === 0 ? a : greatestCommonDivisor(b, a % b);
}

export function toAspectRatioToken(width: number, height: number): string | null {
	if (!Number.isFinite(width) || !Number.isFinite(height) || width <= 0 || height <= 0) {
		return null;
	}
	const w = Math.round(width);
	const h = Math.round(height);
	if (w <= 0 || h <= 0) return null;
	const divisor = greatestCommonDivisor(w, h) || 1;
	return `${w / divisor}:${h / divisor}`;
}

export function anchorRegionsWithDerivedMs(..._args: unknown[]): never {
	throw new Error("pre-v7 OpenScreen documents are not supported");
}
```

```bash
#!/usr/bin/env bash
# Regenerate src/vendor/openscreen-schema.ts from an OpenScreen release tag.
set -euo pipefail
TAG="${1:-v1.12.2}"
URL="https://raw.githubusercontent.com/getopenscreen/openscreen/$TAG/src/lib/ai-edition/schema/index.ts"
OUT="src/vendor/openscreen-schema.ts"
mkdir -p src/vendor
{
	echo "// GENERATED by scripts/vendor-schema.sh from getopenscreen/openscreen@$TAG"
	echo "// (src/lib/ai-edition/schema/index.ts). Do not edit; re-run the script instead."
	curl -fsSL "$URL" | sed \
		-e 's#from "../../../utils/aspectRatioUtils"#from "./shims"#' \
		-e 's#from "../document/ids"#from "./shims"#' \
		-e 's#from "../timeline/timelineMap"#from "./shims"#'
} >"$OUT"
count=$(grep -c 'from "./shims"' "$OUT" || true)
if [ "$count" -ne 3 ]; then
	echo "expected 3 shim imports, found $count — upstream imports changed" >&2
	exit 1
fi
echo "wrote $OUT from $TAG"
```

```ts
import { homedir } from "node:os";
import { join } from "node:path";

export interface Paths {
	dataDir: string;
	projectsDir: string;
	backupsDir: string;
	cacheDir: string;
	whisperModel: string;
}

export function resolvePaths(env: Record<string, string | undefined> = process.env): Paths {
	const dataDir =
		env.OPENSCREEN_DATA_DIR ?? join(homedir(), "Library", "Application Support", "openscreen");
	return {
		dataDir,
		projectsDir: join(dataDir, "projects"),
		backupsDir: join(dataDir, "mcp-backups"),
		cacheDir: join(dataDir, "mcp-cache"),
		whisperModel:
			env.OPENSCREEN_WHISPER_MODEL ??
			join(dataDir, "stt-models", "whisper-ggml", "ggml-small-q8_0.bin"),
	};
}
```

```ts
export type AppCheck = () => Promise<boolean>;

/** True while the OpenScreen app (process name exactly "Openscreen") is running. */
export const isOpenscreenRunning: AppCheck = () =>
	new Promise((resolve, reject) => {
		execFile("/usr/bin/pgrep", ["-x", "Openscreen"], (error) => {
			if (!error) resolve(true);
			else if (error.code === 1) resolve(false);
			else reject(error);
		});
	});
```

```ts
import { ToolError } from "../errors";
import type { Paths } from "../paths";
import { documentSchema } from "../vendor/openscreen-schema";
import type { AppCheck } from "./guard";
import type { Doc } from "./types";

export const APP_OPEN_MESSAGE =
	"OpenScreen is open. It keeps the project in memory and would overwrite this edit on its next save. Quit OpenScreen (⌘Q), then retry.";
const MAX_BACKUPS = 20;
const ID_PATTERN = /^[A-Za-z0-9_-]+$/;

export interface LoadedProject {
	id: string;
	path: string;
	sha256: string;
	doc: Doc;
}

export interface ProjectSummary {
	id: string;
	title: string;
	updatedAt: string;
	durationSec: number;
	counts: { zooms: number; cuts: number; texts: number; blurs: number; speeds: number };
}

export interface Store {
	list(): Promise<ProjectSummary[]>;
	load(ref: string): Promise<LoadedProject>;
	save(loaded: LoadedProject, next: Doc): Promise<LoadedProject>;
	undo(ref: string): Promise<LoadedProject>;
}

interface JournalEntry {
	backup: string;
	writtenSha256: string;
	at: string;
}

const sha256 = (text: string): string => createHash("sha256").update(text).digest("hex");

function formatIssues(error: z.ZodError): string {
	return error.issues
		.slice(0, 5)
		.map((issue) => `${issue.path.join(".") || "(root)"}: ${issue.message}`)
		.join("; ");
}

function summarize(id: string, doc: Doc): ProjectSummary {
	const clips = doc.timeline?.clips ?? [];
	const annotations = doc.annotations ?? [];
	const speeds = doc.legacyEditor?.speedRegions;
	return {
		id,
		title: doc.project?.title ?? id,
		updatedAt: doc.project?.updatedAt ?? "",
		durationSec: clips.reduce((max, c) => Math.max(max, c.timelineEndSec), 0),
		counts: {
			zooms: doc.zoomRanges?.length ?? 0,
			cuts: doc.timeline?.trimRanges?.length ?? 0,
			texts: annotations.filter((a) => a.type === "text").length,
			blurs: annotations.filter((a) => a.type === "blur").length,
			speeds: Array.isArray(speeds) ? speeds.length : 0,
		},
	};
}

async function atomicWrite(path: string, text: string): Promise<void> {
	const tmp = `${path}.${process.pid}.${randomUUID()}.tmp`;
	await writeFile(tmp, text);
	await rename(tmp, path);
}

async function readJournal(dir: string): Promise<JournalEntry[]> {
	try {
		return JSON.parse(await readFile(join(dir, "journal.json"), "utf8")) as JournalEntry[];
	} catch {
		return [];
	}
}

async function writeJournal(dir: string, journal: JournalEntry[]): Promise<void> {
	await atomicWrite(join(dir, "journal.json"), JSON.stringify(journal, null, 2));
}

export function createStore(
	paths: Paths,
	isAppRunning: AppCheck,
	now: () => Date = () => new Date(),
): Store {
	async function list(): Promise<ProjectSummary[]> {
		let names: string[];
		try {
			names = await readdir(paths.projectsDir);
		} catch {
			return [];
		}
		const out: ProjectSummary[] = [];
		for (const name of names.filter((n) => n.endsWith(".openscreen"))) {
			try {
				const doc = JSON.parse(await readFile(join(paths.projectsDir, name), "utf8")) as Doc;
				out.push(summarize(name.replace(/\.openscreen$/, ""), doc));
			} catch {
				// Unreadable or half-written file: leave it out of the list.
			}
		}
		return out.sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
	}

	async function resolveId(ref: string): Promise<string> {
		if (ref === "latest") {
			const [first] = await list();
			if (!first) throw new ToolError(`No OpenScreen projects found in ${paths.projectsDir}.`);
			return first.id;
		}
		if (!ID_PATTERN.test(ref)) {
			throw new ToolError(`"${ref}" is not a valid project id. Use list_projects.`);
		}
		return ref;
	}

	async function load(ref: string): Promise<LoadedProject> {
		const id = await resolveId(ref);
		const path = join(paths.projectsDir, `${id}.openscreen`);
		let text: string;
		try {
			text = await readFile(path, "utf8");
		} catch {
			throw new ToolError(`Project ${id} not found. Use list_projects.`);
		}
		const raw = JSON.parse(text) as { schemaVersion?: unknown };
		if (raw.schemaVersion !== 7) {
			throw new ToolError(
				`Project ${id} uses schemaVersion ${String(raw.schemaVersion)}; this server supports only 7 (OpenScreen 1.12.x). Open and save it in OpenScreen first.`,
			);
		}
		const parsed = documentSchema.safeParse(raw);
		if (!parsed.success) {
			throw new ToolError(
				`Project ${id} does not match the OpenScreen v7 format: ${formatIssues(parsed.error)}`,
			);
		}
		return { id, path, sha256: sha256(text), doc: raw as Doc };
	}

	async function save(loaded: LoadedProject, next: Doc): Promise<LoadedProject> {
		if (await isAppRunning()) throw new ToolError(APP_OPEN_MESSAGE);
		const current = await readFile(loaded.path, "utf8");
		if (sha256(current) !== loaded.sha256) {
			throw new ToolError("The project changed on disk since it was read. Retry the edit.");
		}
		const doc = structuredClone(next);
		doc.project.updatedAt = now().toISOString();
		const parsed = documentSchema.safeParse(doc);
		if (!parsed.success) {
			throw new ToolError(
				`Edit rejected: the result would not be a valid OpenScreen project (${formatIssues(parsed.error)}). Nothing was written.`,
			);
		}
		const text = JSON.stringify(doc, null, 2);
		const dir = join(paths.backupsDir, loaded.id);
		await mkdir(dir, { recursive: true });
		const backup = `${now().toISOString().replace(/[:.]/g, "-")}-${randomUUID().slice(0, 8)}.openscreen`;
		await copyFile(loaded.path, join(dir, backup));
		const journal = await readJournal(dir);
		journal.push({ backup, writtenSha256: sha256(text), at: now().toISOString() });
		while (journal.length > MAX_BACKUPS) {
			const oldest = journal.shift();
			if (oldest) await rm(join(dir, oldest.backup), { force: true });
		}
		await atomicWrite(loaded.path, text);
		await writeJournal(dir, journal);
		return { ...loaded, sha256: sha256(text), doc };
	}

	async function undo(ref: string): Promise<LoadedProject> {
		if (await isAppRunning()) throw new ToolError(APP_OPEN_MESSAGE);
		const loaded = await load(ref);
		const dir = join(paths.backupsDir, loaded.id);
		const journal = await readJournal(dir);
		const last = journal.at(-1);
		if (!last) throw new ToolError("There is no edit made through this server to undo for this project.");
		if (last.writtenSha256 !== loaded.sha256) {
			throw new ToolError(
				"The project changed after the last edit made through this server (probably in the OpenScreen app), so undo would discard that work. Undo it in the app instead.",
			);
		}
		await atomicWrite(loaded.path, await readFile(join(dir, last.backup), "utf8"));
		journal.pop();
		await writeJournal(dir, journal);
		return load(loaded.id);
	}

	return { list, load, save, undo };
}
```

```ts
// Timeline seconds (the RAW ruler in the OpenScreen editor, where cut sections still
// occupy space) ↔ clip-anchored source seconds. Inside one clip the two differ only by
// a shift: raw = clip.timelineStartSec + (source − clip.sourceStartSec).
import { ToolError } from "../errors";
import type { TrimRange } from "../project/types";

export interface ClipLike {
	id: string;
	assetId: string;
	sourceStartSec: number;
	sourceEndSec?: number;
	timelineStartSec: number;
	timelineEndSec: number;
}

export interface Span {
	start: number;
	end: number;
}

export interface Fragment {
	clipId: string;
	assetId: string;
	sourceStartSec: number;
	sourceEndSec: number;
	startMs: number;
	endMs: number;
}

export interface Anchored {
	clipId?: string;
	sourceStartSec?: number;
	sourceEndSec?: number;
	startMs: number;
	endMs: number;
}

const EPS = 1e-6;

export const round3 = (n: number): number => Math.round(n * 1000) / 1000;

export function clipSourceEnd(clip: ClipLike): number {
	return clip.sourceEndSec ?? clip.sourceStartSec + (clip.timelineEndSec - clip.timelineStartSec);
}

export function timelineDurationSec(clips: readonly ClipLike[]): number {
	return clips.reduce((max, c) => Math.max(max, c.timelineEndSec), 0);
}

function byTimeline<C extends ClipLike>(clips: readonly C[]): C[] {
	return [...clips].sort((a, b) => a.timelineStartSec - b.timelineStartSec);
}

function toTimeline(clip: ClipLike, sourceSec: number): number {
	return round3(clip.timelineStartSec + sourceSec - clip.sourceStartSec);
}

/** A timeline span → one anchored fragment per clip it covers. */
export function anchorSpan(clips: readonly ClipLike[], start: number, end: number): Fragment[] {
	if (!(end > start)) {
		throw new ToolError(`end (${end}s) must be after start (${start}s).`);
	}
	const fragments: Fragment[] = [];
	for (const clip of byTimeline(clips)) {
		const a = Math.max(start, clip.timelineStartSec);
		const b = Math.min(end, clip.timelineEndSec);
		if (b - a <= EPS) continue;
		fragments.push({
			clipId: clip.id,
			assetId: clip.assetId,
			sourceStartSec: round3(clip.sourceStartSec + a - clip.timelineStartSec),
			sourceEndSec: round3(clip.sourceStartSec + b - clip.timelineStartSec),
			startMs: Math.round(a * 1000),
			endMs: Math.round(b * 1000),
		});
	}
	if (fragments.length === 0) {
		throw new ToolError(
			`Range ${start}–${end}s is outside the timeline (0–${round3(timelineDurationSec(clips))}s).`,
		);
	}
	return fragments;
}

export function fragmentSpan(fragment: Fragment): Span {
	return { start: fragment.startMs / 1000, end: fragment.endMs / 1000 };
}

/** A source time of one asset → its timeline time on every clip that plays it. */
export function sourceToTimeline(
	clips: readonly ClipLike[],
	assetId: string,
	sourceSec: number,
): number[] {
	return byTimeline(clips)
		.filter(
			(c) =>
				c.assetId === assetId &&
				sourceSec >= c.sourceStartSec - EPS &&
				sourceSec <= clipSourceEnd(c) + EPS,
		)
		.map((c) => toTimeline(c, sourceSec));
}

/** A source span of one asset → its timeline span on every clip that plays part of it. */
export function sourceSpanToTimeline(
	clips: readonly ClipLike[],
	assetId: string,
	start: number,
	end: number,
): Span[] {
	const spans: Span[] = [];
	for (const clip of byTimeline(clips)) {
		if (clip.assetId !== assetId) continue;
		const lo = Math.max(start, clip.sourceStartSec);
		const hi = Math.min(end, clipSourceEnd(clip));
		if (hi - lo > EPS) spans.push({ start: toTimeline(clip, lo), end: toTimeline(clip, hi) });
	}
	return spans;
}

/** A timeline time → the clip playing it and the source time within that clip. */
export function timelineToSource<C extends ClipLike>(
	clips: readonly C[],
	t: number,
): { clip: C; sourceSec: number } | null {
	const sorted = byTimeline(clips);
	const last = sorted.at(-1);
	const clip =
		sorted.find((c) => t >= c.timelineStartSec && t < c.timelineEndSec) ??
		(last && Math.abs(t - last.timelineEndSec) < EPS ? last : undefined);
	return clip ? { clip, sourceSec: round3(clip.sourceStartSec + t - clip.timelineStartSec) } : null;
}

/** Where a zoom/annotation/speed fragment sits on the timeline. The anchor wins over the ms cache. */
export function regionSpanSec(clips: readonly ClipLike[], region: Anchored): Span {
	const clip = region.clipId ? clips.find((c) => c.id === region.clipId) : undefined;
	if (clip && region.sourceStartSec !== undefined && region.sourceEndSec !== undefined) {
		return { start: toTimeline(clip, region.sourceStartSec), end: toTimeline(clip, region.sourceEndSec) };
	}
	return { start: region.startMs / 1000, end: region.endMs / 1000 };
}

/** Where a cut sits on the timeline, or null when its clip is gone. */
export function trimSpanSec(
	clips: readonly ClipLike[],
	trim: Pick<TrimRange, "assetId" | "clipId" | "startSec" | "endSec">,
): Span | null {
	const clip = trim.clipId
		? clips.find((c) => c.id === trim.clipId)
		: byTimeline(clips).find(
				(c) =>
					c.assetId === trim.assetId &&
					trim.startSec >= c.sourceStartSec &&
					trim.startSec <= clipSourceEnd(c),
			);
	if (!clip) return null;
	const lo = Math.max(trim.startSec, clip.sourceStartSec);
	const hi = Math.min(trim.endSec, clipSourceEnd(clip));
	if (hi - lo <= EPS) return null;
	return { start: toTimeline(clip, lo), end: toTimeline(clip, hi) };
}
```

```ts
import { ToolError } from "../errors";
import type { Doc, SpeedRegion } from "../project/types";
import { regionSpanSec, type Span, trimSpanSec } from "./timemap";

export type RegionKind = "zoom" | "text" | "blur" | "speed" | "cut";

export interface RegionRef {
	kind: RegionKind;
	id: string;
	span: Span | null;
}

export interface EditResult {
	doc: Doc;
	ids: string[];
}

export function speedRegions(doc: Doc): SpeedRegion[] {
	const list = doc.legacyEditor?.speedRegions;
	return Array.isArray(list) ? (list as SpeedRegion[]) : [];
}

export function allRegions(doc: Doc): RegionRef[] {
	const clips = doc.timeline.clips;
	return [
		...doc.zoomRanges.map((z) => ({ kind: "zoom" as const, id: z.id, span: regionSpanSec(clips, z) })),
		...doc.annotations
			.filter((a) => a.type === "text" || a.type === "blur")
			.map((a) => ({ kind: a.type as "text" | "blur", id: a.id, span: regionSpanSec(clips, a) })),
		...speedRegions(doc).map((s) => ({ kind: "speed" as const, id: s.id, span: regionSpanSec(clips, s) })),
		...doc.timeline.trimRanges.map((t) => ({ kind: "cut" as const, id: t.id, span: trimSpanSec(clips, t) })),
	];
}

export function findRegion(doc: Doc, id: string): RegionRef | null {
	return allRegions(doc).find((r) => r.id === id) ?? null;
}

export function fmtSpan(span: Span): string {
	return `${span.start.toFixed(2)}–${span.end.toFixed(2)}s`;
}

export function regionNotFound(id: string): ToolError {
	return new ToolError(`No zoom, cut, text, blur or speed region has id ${id}. Use describe_project.`);
}

const EPS = 1e-6;

/** Repel rule: two regions of the same kind may not overlap (touching is fine). */
export function assertNoOverlap(
	doc: Doc,
	kind: Exclude<RegionKind, "cut">,
	spans: Span[],
	ignoreId?: string,
): void {
	const others = allRegions(doc).filter((r) => r.kind === kind && r.id !== ignoreId && r.span);
	for (const span of spans) {
		for (const other of others) {
			const o = other.span as Span;
			if (span.start < o.end - EPS && o.start < span.end - EPS) {
				throw new ToolError(
					`That ${kind} (${fmtSpan(span)}) overlaps ${kind} ${other.id} (${fmtSpan(o)}). Update or remove ${other.id} first.`,
				);
			}
		}
	}
}

export function removeRegion(doc: Doc, id: string): EditResult {
	const ref = findRegion(doc, id);
	if (!ref) throw regionNotFound(id);
	const next = structuredClone(doc);
	switch (ref.kind) {
		case "zoom":
			next.zoomRanges = next.zoomRanges.filter((z) => z.id !== id);
			break;
		case "text":
		case "blur":
			next.annotations = next.annotations.filter((a) => a.id !== id);
			break;
		case "speed":
			(next.legacyEditor as Record<string, unknown>).speedRegions = speedRegions(next).filter((s) => s.id !== id);
			break;
		case "cut":
			next.timeline.trimRanges = next.timeline.trimRanges.filter((t) => t.id !== id);
			break;
	}
	return { doc: next, ids: [id] };
}
```

```ts
import { ToolError } from "../errors";
import type { Doc, SpeedRegion } from "../project/types";
import { createId } from "../vendor/shims";
import { assertNoOverlap, type EditResult } from "./regions";
import { anchorSpan, fragmentSpan } from "./timemap";

export const MIN_SPEED = 0.1;
// Above 16× Chromium cannot play natively and OpenScreen's preview falls back to frame-stepping.
export const MAX_SPEED = 16;

export function checkSpeed(speed: number): number {
	if (!(speed >= MIN_SPEED && speed <= MAX_SPEED)) {
		throw new ToolError(`speed must be between ${MIN_SPEED} and ${MAX_SPEED} (got ${speed}).`);
	}
	return Math.round(speed * 100) / 100;
}

export interface SpeedInput {
	start: number;
	end: number;
	speed: number;
}

export function setSpeed(doc: Doc, input: SpeedInput): EditResult {
	const speed = checkSpeed(input.speed);
	const fragments = anchorSpan(doc.timeline.clips, input.start, input.end);
	assertNoOverlap(doc, "speed", fragments.map(fragmentSpan));
	const next = structuredClone(doc);
	const legacy = (next.legacyEditor ??= {});
	if (!Array.isArray(legacy.speedRegions)) legacy.speedRegions = [];
	const list = legacy.speedRegions as SpeedRegion[];
	const ids = fragments.map((f) => {
		const id = createId("speed");
		list.push({
			id,
			startMs: f.startMs,
			endMs: f.endMs,
			speed,
			clipId: f.clipId,
			sourceStartSec: f.sourceStartSec,
			sourceEndSec: f.sourceEndSec,
		});
		return id;
	});
	return { doc: next, ids };
}
```

```ts
import { ToolError } from "../errors";
import type { Doc } from "../project/types";
import { assertNoOverlap, type EditResult, findRegion, regionNotFound, speedRegions } from "./regions";
import { checkSpeed } from "./speed";
import { anchorSpan, fragmentSpan, round3 } from "./timemap";
import type { CursorFocus, Depth, Focus } from "./zoom";

export interface UpdateInput {
	id: string;
	start?: number;
	end?: number;
	depth?: Depth;
	focus?: Focus;
	text?: string;
	fontSize?: number;
	color?: string;
	speed?: number;
	reason?: string;
}

/** Patches one region fragment in place, so fields this server does not know survive. */
export function updateRegion(doc: Doc, input: UpdateInput, cursorFocus: CursorFocus): EditResult {
	const ref = findRegion(doc, input.id);
	if (!ref) throw regionNotFound(input.id);
	if (!ref.span) {
		throw new ToolError(`Region ${input.id} is no longer on the timeline (its clip was removed). Remove it instead.`);
	}
	const fragments = anchorSpan(doc.timeline.clips, input.start ?? ref.span.start, input.end ?? ref.span.end);
	if (fragments.length !== 1) {
		throw new ToolError("An update cannot make a region cross a clip boundary. Remove it and add it again instead.");
	}
	const f = fragments[0]!;
	if (ref.kind !== "cut") assertNoOverlap(doc, ref.kind, [fragmentSpan(f)], input.id);
	const anchor = {
		startMs: f.startMs,
		endMs: f.endMs,
		clipId: f.clipId,
		sourceStartSec: f.sourceStartSec,
		sourceEndSec: f.sourceEndSec,
	};
	const next = structuredClone(doc);
	switch (ref.kind) {
		case "zoom": {
			const zoom = next.zoomRanges.find((z) => z.id === input.id)!;
			Object.assign(zoom, anchor);
			if (input.depth) zoom.depth = input.depth;
			if (input.focus === "cursor") {
				const focus = cursorFocus(f.assetId, f.sourceStartSec, f.sourceEndSec) ?? zoom.focus;
				zoom.focus = { cx: round3(focus.cx), cy: round3(focus.cy) };
				zoom.focusMode = "auto";
			} else if (input.focus) {
				zoom.focus = { cx: round3(input.focus.cx), cy: round3(input.focus.cy) };
				zoom.focusMode = "manual";
			}
			break;
		}
		case "text":
		case "blur": {
			const annotation = next.annotations.find((a) => a.id === input.id)!;
			Object.assign(annotation, anchor);
			if (input.text !== undefined && ref.kind === "text") {
				annotation.content = input.text;
				annotation.textContent = input.text;
			}
			if (input.fontSize !== undefined) annotation.style.fontSize = input.fontSize;
			if (input.color !== undefined) annotation.style.color = input.color;
			break;
		}
		case "speed": {
			const region = speedRegions(next).find((s) => s.id === input.id)!;
			Object.assign(region, anchor);
			if (input.speed !== undefined) region.speed = checkSpeed(input.speed);
			break;
		}
		case "cut": {
			const trim = next.timeline.trimRanges.find((t) => t.id === input.id)!;
			Object.assign(trim, {
				assetId: f.assetId,
				clipId: f.clipId,
				startSec: f.sourceStartSec,
				endSec: f.sourceEndSec,
			});
			if (input.reason !== undefined) trim.reason = input.reason;
			break;
		}
	}
	return { doc: next, ids: [input.id] };
}
```

```ts
import { basename } from "node:path";
import { speedRegions } from "./edit/regions";
import {
	clipSourceEnd,
	regionSpanSec,
	round3,
	type Span,
	timelineDurationSec,
	trimSpanSec,
} from "./edit/timemap";
import type { Doc } from "./project/types";

export function unionLength(spans: Span[]): number {
	const sorted = [...spans].sort((a, b) => a.start - b.start);
	let total = 0;
	let current: Span | null = null;
	for (const span of sorted) {
		if (current && span.start <= current.end) {
			current.end = Math.max(current.end, span.end);
		} else {
			if (current) total += current.end - current.start;
			current = { ...span };
		}
	}
	if (current) total += current.end - current.start;
	return round3(total);
}

export function describeProject(doc: Doc) {
	const clips = doc.timeline.clips;
	const durationSec = round3(timelineDurationSec(clips));
	const cuts = doc.timeline.trimRanges.map((t) => {
		const span = trimSpanSec(clips, t);
		return { id: t.id, start: span?.start ?? null, end: span?.end ?? null, reason: t.reason };
	});
	const cutSpans = doc.timeline.trimRanges
		.map((t) => trimSpanSec(clips, t))
		.filter((s): s is Span => s !== null);
	return {
		id: doc.project.id,
		title: doc.project.title,
		timeUnit: "timeline seconds (cut sections still occupy their place)",
		durationSec,
		keptDurationSec: round3(durationSec - unionLength(cutSpans)),
		clips: clips.map((c) => ({
			id: c.id,
			start: c.timelineStartSec,
			end: c.timelineEndSec,
			sourceStart: c.sourceStartSec,
			sourceEnd: clipSourceEnd(c),
			video: basename(doc.assets.find((a) => a.id === c.assetId)?.originalPath ?? "?"),
		})),
		cuts,
		zooms: doc.zoomRanges.map((z) => ({
			id: z.id,
			...regionSpanSec(clips, z),
			depth: z.depth,
			focus: z.focus,
			focusMode: z.focusMode ?? "manual",
		})),
		texts: doc.annotations
			.filter((a) => a.type === "text")
			.map((a) => ({ id: a.id, ...regionSpanSec(clips, a), text: a.textContent ?? a.content })),
		blurs: doc.annotations
			.filter((a) => a.type === "blur")
			.map((a) => ({
				id: a.id,
				...regionSpanSec(clips, a),
				rect: { ...a.position, ...a.size },
				style: a.blurData?.type ?? "mosaic",
			})),
		speeds: speedRegions(doc).map((s) => ({ id: s.id, ...regionSpanSec(clips, s), speed: s.speed })),
```

```ts
// OpenScreen writes `<video>.cursor.json` next to every recording:
// { version: 2, samples: [{ timeMs, cx, cy, visible, interactionType: "move" | "click" | "mouseup" }] }
// timeMs is the recording's source time; cx/cy are 0–1 fractions of the frame.
// Samples arrive only when something happens, so a gap between samples means the cursor held still.
import { readFile } from "node:fs/promises";
import { round3 } from "../edit/timemap";

export interface CursorSample {
	timeMs: number;
	cx: number;
	cy: number;
	interactionType: string;
}

export interface Click {
	sec: number;
	cx: number;
	cy: number;
}

export interface Stillness {
	startSec: number;
	endSec: number;
	cx: number;
	cy: number;
	kind: "dwell" | "idle";
}

/** "Still" = moved less than 0.5 % of the frame. */
export const STILL_TOLERANCE = 0.005;
export const MIN_DWELL_SEC = 1.5;
export const MIN_IDLE_SEC = 2;

export async function loadCursorSamples(videoPath: string): Promise<CursorSample[] | null> {
	try {
		const raw = JSON.parse(await readFile(`${videoPath}.cursor.json`, "utf8")) as {
			samples?: CursorSample[];
		};
		return (raw.samples ?? [])
			.filter((s) => Number.isFinite(s.timeMs) && Number.isFinite(s.cx) && Number.isFinite(s.cy))
			.sort((a, b) => a.timeMs - b.timeMs);
	} catch {
		return null;
	}
}

export function findClicks(samples: readonly CursorSample[]): Click[] {
	return samples
		.filter((s) => s.interactionType === "click")
		.map((s) => ({ sec: round3(s.timeMs / 1000), cx: round3(s.cx), cy: round3(s.cy) }));
}

/** Stretches of ≥ 1.5 s where the cursor stays put. Click-free stretches of ≥ 2 s are "idle". */
export function findStillStretches(samples: readonly CursorSample[], endMs: number): Stillness[] {
	const out: Stillness[] = [];
	const first = samples[0];
	if (!first) return out;
	const clicks = findClicks(samples).map((c) => c.sec);
	let anchor = first;
	const close = (stretchEndMs: number) => {
		const startSec = anchor.timeMs / 1000;
		const endSec = stretchEndMs / 1000;
		const length = endSec - startSec;
		if (length < MIN_DWELL_SEC) return;
		const hasClick = clicks.some((c) => c >= startSec - 1e-6 && c <= endSec + 1e-6);
		out.push({
			startSec: round3(startSec),
			endSec: round3(endSec),
			cx: round3(anchor.cx),
			cy: round3(anchor.cy),
			kind: !hasClick && length >= MIN_IDLE_SEC ? "idle" : "dwell",
		});
	};
	for (const sample of samples.slice(1)) {
		const moved = Math.max(Math.abs(sample.cx - anchor.cx), Math.abs(sample.cy - anchor.cy));
		if (moved >= STILL_TOLERANCE) {
			close(sample.timeMs);
			anchor = sample;
		}
	}
	close(Math.max(endMs, anchor.timeMs));
	return out;
}

function median(values: number[]): number {
	const sorted = [...values].sort((a, b) => a - b);
	const mid = Math.floor(sorted.length / 2);
	return sorted.length % 2 ? sorted[mid]! : (sorted[mid - 1]! + sorted[mid]!) / 2;
}

export function medianPosition(
	samples: readonly CursorSample[],
	startSec: number,
	endSec: number,
): { cx: number; cy: number } | null {
	const inside = samples.filter((s) => s.timeMs >= startSec * 1000 && s.timeMs <= endSec * 1000);
	if (inside.length === 0) return null;
	return { cx: round3(median(inside.map((s) => s.cx))), cy: round3(median(inside.map((s) => s.cy))) };
}
```

```ts
		try {
			await access(candidate, constants.X_OK);
			return candidate;
		} catch {
			// not here
		}
	}
	return null;
}

export async function findBins(env: Record<string, string | undefined> = process.env): Promise<Bins> {

		ffprobe: await findBinary("ffprobe", env),

	};
}

export function requireBin(path: string | null, name: string, hint: string): string {
	if (!path) throw new ToolError(`${name} not found. ${hint}`);
	return path;

export function run(
	bin: string,
	args: string[],
	opts: { timeoutMs?: number } = {},
): Promise<{ stdout: Buffer; stderr: string }> {
	return new Promise((resolve, reject) => {
		execFile(
			bin,
			args,
			{ encoding: "buffer", maxBuffer: 256 * 1024 * 1024, timeout: opts.timeoutMs ?? 120_000 },
			(error, stdout, stderr) => {
				const errText = stderr.toString();
				if (error) {
					const tail = errText.trim().split("\n").slice(-3).join(" ");
					reject(new ToolError(`${basename(bin)} failed: ${tail || error.message}`));
				} else {
					resolve({ stdout, stderr: errText });
				}
			},
		);
	});
}

	]);
	return stdout.toString().trim().length > 0;
}
```

```ts
import { mkdir, readFile, stat, writeFile } from "node:fs/promises";
import { dirname, join } from "node:path";

export interface Cache {
	get<T>(assetId: string, name: string, video: string): Promise<T | null>;
	set<T>(assetId: string, name: string, video: string, value: T): Promise<void>;
}

/** Per-asset JSON cache, invalidated when the video's size or mtime changes. */
export function createCache(dir: string): Cache {
	const file = (assetId: string, name: string) =>
		join(dir, assetId.replace(/[^A-Za-z0-9_-]/g, "_"), `${name}.json`);
	const key = async (video: string) => {
		const s = await stat(video);
		return `${s.size}:${Math.round(s.mtimeMs)}`;
	};
	return {
		async get<T>(assetId: string, name: string, video: string): Promise<T | null> {
			try {
				const stored = JSON.parse(await readFile(file(assetId, name), "utf8")) as { key: string; value: T };
				return stored.key === (await key(video)) ? stored.value : null;
			} catch {
				return null;
			}
		},
		async set<T>(assetId: string, name: string, video: string, value: T): Promise<void> {
			const path = file(assetId, name);
			await mkdir(dirname(path), { recursive: true });
			await writeFile(path, JSON.stringify({ key: await key(video), value }));
		},
	};
}
```

```ts
	language: string;

interface WhisperJson {
	result?: { language?: string };
	transcription?: { offsets: { from: number; to: number }; text: string }[];
}

export function parseWhisperJson(json: unknown): Transcript {
	const data = json as WhisperJson;
	return {
		language: data.result?.language ?? "unknown",
		segments: (data.transcription ?? [])
			.map((s) => ({
				startSec: round3(s.offsets.from / 1000),
				endSec: round3(s.offsets.to / 1000),
				text: s.text.trim(),
			}))
			.filter((s) => s.text.length > 0 && s.text !== "[BLANK_AUDIO]"),
	};
}
```

```ts
function assetsOnTimeline(doc: Doc): AssetOnTimeline[] {

async function cachedScenes(ctx: ToolContext, asset: AssetOnTimeline, ffmpeg: string): Promise<number[]> {
	const cached = await ctx.cache.get<number[]>(asset.assetId, "scenes", asset.video);
	if (cached) return cached;
	const scenes = await detectScenes(ffmpeg, asset.video);
	await ctx.cache.set(asset.assetId, "scenes", asset.video, scenes);

export async function analyzeDoc(ctx: ToolContext, doc: Doc, include: readonly Signal[]): Promise<Analysis> {
	const clips = doc.timeline.clips;
	const events: TimelineEvent[] = [];
	const notes: string[] = [];
	for (const asset of assetsOnTimeline(doc)) {
		const name = basename(asset.video);
		const exists = await fileExists(asset.video);
		if (!exists) notes.push(`Video missing: ${asset.video}. Frames, scenes and transcript are unavailable for it.`);

		if (include.includes("cursor")) {
			const samples = await loadCursorSamples(asset.video);
			if (!samples) {
				notes.push(`No cursor data for ${name}.`);
			} else {
				const endMs = Math.max(asset.endMs, samples.at(-1)?.timeMs ?? 0);
				for (const click of findClicks(samples)) {
					for (const at of sourceToTimeline(clips, asset.assetId, click.sec)) {
						events.push({ at, kind: "click", cx: click.cx, cy: click.cy });
					}
				}
				for (const still of findStillStretches(samples, endMs)) {
					for (const span of sourceSpanToTimeline(clips, asset.assetId, still.startSec, still.endSec)) {
						events.push({ at: span.start, end: span.end, kind: still.kind, cx: still.cx, cy: still.cy });
					}
				}
			}
		}

		if (include.includes("scenes") && exists) {
			if (!ctx.bins.ffmpeg) {
				notes.push(`ffmpeg not found, so scene changes were skipped. ${FFMPEG_HINT}`);
			} else {
				for (const sec of await cachedScenes(ctx, asset, ctx.bins.ffmpeg)) {
					for (const at of sourceToTimeline(clips, asset.assetId, sec)) events.push({ at, kind: "scene" });
				}
			}
		}

		if (include.includes("transcript")) {
			const transcript = exists
				? await ctx.cache.get<Transcript>(asset.assetId, "transcript", asset.video)
				: null;
			if (!transcript) {
				notes.push(`No transcript for ${name} yet; call transcribe first.`);
			} else {
				for (const seg of transcript.segments) {
					for (const span of sourceSpanToTimeline(clips, asset.assetId, seg.startSec, seg.endSec)) {
						events.push({ at: span.start, end: span.end, kind: "speech", text: seg.text });
					}
				}
			}
		}
	}
	events.sort((a, b) => a.at - b.at);
	return {
		timeUnit: "timeline seconds",
		durationSec: round3(timelineDurationSec(clips)),
		events,
		notes,
	};
}

export async function analyze(ctx: ToolContext, ref: string, include: readonly Signal[]): Promise<Analysis> {
	return analyzeDoc(ctx, (await ctx.store.load(ref)).doc, include);
}

const pos = (e: TimelineEvent) => `(${e.cx}, ${e.cy})`;

function describeEvent(e: TimelineEvent): string {
	switch (e.kind) {
		case "click":
			return `click at ${pos(e)}`;
		case "dwell":
			return `cursor rests at ${pos(e)} until ${e.end?.toFixed(2)}s`;
		case "idle":
			return `idle until ${e.end?.toFixed(2)}s`;
		case "scene":
			return "scene change";
		case "speech":
			return `says "${e.text}"`;
	}
}

function pickEvenly(values: number[], max: number): number[] {
	if (values.length <= max) return values;
	if (max === 1) return [values[0]!];
	return Array.from({ length: max }, (_, i) => values[Math.round((i * (values.length - 1)) / (max - 1))]!);
}

export async function frames(
	ctx: ToolContext,
	ref: string,
	opts: { times?: number[]; at?: "clicks" | "scenes" | "evenly"; max: number },
): Promise<Frame[]> {
	const ffmpeg = requireBin(ctx.bins.ffmpeg, "ffmpeg", FFMPEG_HINT);
	const { doc } = await ctx.store.load(ref);
	const clips = doc.timeline.clips;
	const duration = timelineDurationSec(clips);
	const at = opts.times?.length ? undefined : (opts.at ?? "evenly");
	const analysis = await analyzeDoc(ctx, doc, at === "scenes" ? ["cursor", "scenes"] : ["cursor"]);

	let times: number[];
	if (opts.times?.length) times = opts.times;
	else if (at === "clicks") times = analysis.events.filter((e) => e.kind === "click").map((e) => e.at);
	else if (at === "scenes") times = [0, ...analysis.events.filter((e) => e.kind === "scene").map((e) => e.at)];
	else times = Array.from({ length: opts.max }, (_, i) => round3(((i + 0.5) * duration) / opts.max));
	times = pickEvenly([...new Set(times.map(round3))].sort((a, b) => a - b), opts.max);
	if (times.length === 0) throw new ToolError(`No ${at} found in this recording.`);

	const out: Frame[] = [];
	for (const t of times) {
		const position = timelineToSource(clips, t);
		if (!position) throw new ToolError(`${t}s is outside the timeline (0–${round3(duration)}s).`);
		const video = doc.assets.find((a) => a.id === position.clip.assetId)?.originalPath;
		if (!video || !(await fileExists(video))) throw new ToolError(`The video for ${t}s is missing.`);
		const nearby = analysis.events.filter(
			(e) => Math.abs(e.at - t) <= 0.5 || (e.end !== undefined && e.at <= t && t <= e.end),
		);
		const caption = `t=${t.toFixed(2)}s${nearby.length ? `: ${nearby.map(describeEvent).join("; ")}` : ""}`;
		out.push({ at: t, caption, jpeg: await extractFrame(ffmpeg, video, position.sourceSec) });
	}
	return out;
}

export async function transcribe(ctx: ToolContext, ref: string) {
	const ffmpeg = requireBin(ctx.bins.ffmpeg, "ffmpeg", FFMPEG_HINT);
	const whisper = requireBin(ctx.bins.whisper, "whisper-cli", "Install it with `brew install whisper-cpp`.");
	if (!(await fileExists(ctx.paths.whisperModel))) {
		throw new ToolError(
			`Whisper model not found at ${ctx.paths.whisperModel}. Turn on captions once in OpenScreen to download it.`,
		);
	}
	const { doc } = await ctx.store.load(ref);
	const results: { video: string; status: string; language?: string; segments?: number }[] = [];
	for (const asset of assetsOnTimeline(doc)) {
		const video = basename(asset.video);
		if (!(await fileExists(asset.video))) {
			results.push({ video, status: "video missing" });
			continue;
		}
		const cached = await ctx.cache.get<Transcript>(asset.assetId, "transcript", asset.video);
		if (cached) {
			results.push({ video, status: "already transcribed", language: cached.language, segments: cached.segments.length });
			continue;
		}
		if (ctx.bins.ffprobe && !(await hasAudio(ctx.bins.ffprobe, asset.video))) {
			results.push({ video, status: "no audio stream" });
			continue;
		}
		const transcript = await transcribeVideo({
			ffmpeg,
			whisper,
			model: ctx.paths.whisperModel,
			video: asset.video,
			workDir: join(ctx.paths.cacheDir, "tmp", randomUUID()),
		});
		await ctx.cache.set(asset.assetId, "transcript", asset.video, transcript);
		results.push({ video, status: "transcribed", language: transcript.language, segments: transcript.segments.length });
	}
	return results;
}
```

```ts
import { medianPosition } from "../analysis/cursor";
import { describeProject, type ProjectDescription } from "../describe";
import type { EditResult } from "../edit/regions";
import type { CursorFocus } from "../edit/zoom";
import type { Doc } from "../project/types";
import type { ToolContext } from "./context";
import { cursorSamplesByAsset } from "./understand";

const REMINDER = "Saved. Reopen this project in OpenScreen to preview the change and export.";

export async function applyEdit(
	ctx: ToolContext,
	ref: string,
	edit: (doc: Doc, cursorFocus: CursorFocus) => EditResult,
): Promise<{ changedIds: string[]; project: ProjectDescription; reminder: string }> {
	const loaded = await ctx.store.load(ref);
	const samples = await cursorSamplesByAsset(loaded.doc);
	const cursorFocus: CursorFocus = (assetId, start, end) =>
		medianPosition(samples.get(assetId) ?? [], start, end);
	const { doc, ids } = edit(loaded.doc, cursorFocus);
	const saved = await ctx.store.save(loaded, doc);
	return { changedIds: ids, project: describeProject(saved.doc), reminder: REMINDER };
}

export async function undoEdit(ctx: ToolContext, ref: string) {
	const restored = await ctx.store.undo(ref);
	return { undone: true as const, project: describeProject(restored.doc), reminder: REMINDER };
}
```

```ts
All times are timeline seconds as shown in the OpenScreen editor; cut sections keep their place on that timeline, so times from analyze_recording can be passed straight to edit tools.
Edits fail while the OpenScreen app is running: ask the user to quit it (⌘Q). After editing, tell the user to reopen the project in OpenScreen to preview and export.

	project,
	start: sec.describe("Start, in timeline seconds."),

function json(value: unknown): CallToolResult {
	return { content: [{ type: "text", text: JSON.stringify(value, null, 2) }] };
}

function guarded<A>(fn: (args: A) => Promise<CallToolResult>): (args: A) => Promise<CallToolResult> {
	return async (args) => {
		try {
			return await fn(args);
		} catch (error) {
			if (error instanceof ToolError) {
				return { content: [{ type: "text", text: error.message }], isError: true };
			}
			console.error(error);
			return {
				content: [{ type: "text", text: `Unexpected error: ${(error as Error).message}` }],
				isError: true,
			};
		}
	};
}
```
